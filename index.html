  <section class="formula" aria-label="Equation">

    <div class="formula-label">Capital-only product equation</div>

    <div class="formula-equation">i<sub>target,i</sub> = c<sub>i</sub> + <span>(TVOG<sub>i</sub> + E<sub>i</sub> + Strain<sub>i</sub> - U<sub>i</sub>) / L<sub>i</sub></span> + <span>(NA<sub>i</sub> / L<sub>i</sub>) * (R<sub>pre-tax</sub> - i<sub>trad</sub>)</span> + ALM<sub>i</sub></div>

    <div class="formula-note">All inputs are pre-tax annual values. TVOG, expenses, strain, and underwriting profit are amounts in the same unit as L; ALM is a rate. NA<sub>i</sub> is allocated from C-ROSS required capital, not reserve size.</div>

  <section class="panel original-bridge">

    <div class="panel-heading"><div><p class="eyebrow">Keep the PPT logic</p><h2>Original equation, with one C-ROSS improvement</h2></div></div>

    <div class="bridge-grid">

      <div class="bridge-step">

        <span>1. Original PPT equation</span>

        <strong>i<sub>target,i</sub> = c<sub>i</sub> + (NA<sub>i</sub> / L<sub>i</sub>) * (R<sub>pre-tax</sub> - i<sub>trad</sub>)</strong>

        <p>The pricing / target-yield logic remains unchanged.</p>

      </div>

      <div class="bridge-arrow" aria-hidden="true">→</div>

      <div class="bridge-step highlight">

        <span>2. Change only the NA allocation basis</span>

        <strong>Reserve allocation → C-ROSS required-capital allocation</strong>

        <p>Required capital determines each product’s share of the net-asset pool.</p>

      </div>

      <div class="bridge-arrow" aria-hidden="true">→</div>

      <div class="bridge-step">

        <span>3. Use the same equation with better NA</span>

        <strong>NA<sub>i</sub> = Pool * RC<sub>i</sub><sup>alloc</sup> / ΣRC<sup>alloc</sup></strong>

        <p>Use this in ALLOC mode so total allocated NA equals the actual pool.</p>

      </div>

    </div>

    <p class="helper"><strong>Do not simply replace “NA” with “RC” in the equation.</strong> RC is the risk measure used to allocate capital; NA remains the actual shareholder capital assigned to the block. In TARGET mode, you can instead use NA<sub>i</sub> = k * RC<sub>i</sub><sup>alloc</sup>.</p>

  </section>

    <h2>What this model proves—and what it does not</h2>

<script>

  const defaults={sarmra:70,k:1.5,roe:.10,tax:.25,trad:.04,pool:50,mode:'ALLOC',im:0,ic:0,mc:.15,blocks:[

    {name:'A',l:300,c:.03,tvog:0,expense:0,strain:0,underwriting:0,alm:0,ins:6,mkt:4,credit:2,other:0},

    {name:'B',l:200,c:.032,tvog:0,expense:0,strain:0,underwriting:0,alm:0,ins:4,mkt:5,credit:3,other:0},

    {name:'C',l:100,c:.035,tvog:0,expense:0,strain:0,underwriting:0,alm:0,ins:5,mkt:14,credit:6,other:0}

  ]};

  let state=JSON.parse(JSON.stringify(defaults));

  let baseline=JSON.parse(JSON.stringify(defaults));

  const $=id=>document.getElementById(id);

  const fmtPct=(v,d=2)=>Number.isFinite(v)?`${(v*100).toFixed(d)}%`:'—';

  const fmtNum=(v,d=2)=>Number.isFinite(v)?Number(v).toLocaleString(undefined,{minimumFractionDigits:d,maximumFractionDigits:d}):'—';

  const num=v=>{const n=parseFloat(v);return Number.isFinite(n)?n:0};

  const clone=x=>JSON.parse(JSON.stringify(x));


  function calculate(input=state){

    const q=.4-.005*input.sarmra;

    const preTax=input.tax<1?input.roe/(1-input.tax):0;

    const rows=input.blocks.map(b=>{

      const rad=b.ins*b.ins+b.mkt*b.mkt+b.credit*b.credit+2*input.im*b.ins*b.mkt+2*input.ic*b.ins*b.credit+2*input.mc*b.mkt*b.credit;

      const quant=Math.sqrt(Math.max(0,rad));

      const control=q*quant;

      const rc=quant+control+b.other;

      return {...b,quant,control,rc};

    });

    const totalRC=rows.reduce((s,r)=>s+r.rc,0);

    rows.forEach(r=>{

      r.na=input.mode==='ALLOC'?(totalRC?input.pool*r.rc/totalRC:0):r.rc*input.k;

      r.intensity=r.l?r.na/r.l:0;

      r.cost=r.intensity*(preTax-input.trad);

      r.otherCost=r.l?(r.tvog+r.expense+r.strain-r.underwriting)/r.l+r.alm:0;

      r.target=r.c+r.otherCost+r.cost;

    });

    const ranked=[...rows].sort((a,b)=>b.target-a.target);

    ranked.forEach((r,i)=>r.rank=i+1);

    return {q,preTax,rows,totalRC,totalNA:rows.reduce((s,r)=>s+r.na,0),targetTotal:input.k*totalRC,freeSurplus:input.pool-input.k*totalRC};

  }

  function normalizeBlocks(){

    state.blocks=state.blocks.map((b,i)=>({name:`Block ${i+1}`,l:0,c:0,tvog:0,expense:0,strain:0,underwriting:0,alm:0,ins:0,mkt:0,credit:0,other:0,...b}));

  }


  function assignInputs(){

    ['sarmra','k','roe','tax','trad','pool'].forEach(id=>$(id).value=state[id]);

    $('mode').value=state.mode;$('corr-im').value=state.im;$('corr-ic').value=state.ic;$('corr-mc').value=state.mc;

  }

  function updateStateFromInputs(){

    ['sarmra','k','roe','tax','trad','pool'].forEach(id=>state[id]=num($(id).value));

    state.mode=$('mode').value;state.im=num($('corr-im').value);state.ic=num($('corr-ic').value);state.mc=num($('corr-mc').value);

  }

  function escapeHtml(value){return String(value).replace(/[&<>'"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[c]));}

  function rankMap(model){return Object.fromEntries(model.rows.map(r=>[r.name,r.rank]));}

  function hasChanged(keys){return keys.some(key=>state[key]!==baseline[key]);}

  function rowsChanged(keys){return state.blocks.length!==baseline.blocks.length||state.blocks.some((r,i)=>!baseline.blocks[i]||keys.some(key=>r[key]!==baseline.blocks[i][key]));}

  function renderMovementMap(model){

    const base=calculate(baseline),baseRanks=rankMap(base),currentRanks=rankMap(model);

    const maxDelta=Math.max(...model.rows.map(r=>{const old=base.rows.find(x=>x.name===r.name);return Math.abs(r.target-(old?old.target:0));}));

    const moved=model.rows.filter(r=>baseRanks[r.name]!==currentRanks[r.name]).length;

    const activate=(id,on)=>$(id).classList.toggle('active',on);

    activate('mover-risk',rowsChanged(['ins','mkt','credit','other']));activate('mover-corr',hasChanged(['im','ic','mc']));

    activate('mover-liability',rowsChanged(['l']));activate('mover-cost',rowsChanged(['c','tvog','expense','strain','underwriting','alm']));

    activate('mover-roe',hasChanged(['roe']));activate('mover-tax',hasChanged(['tax']));activate('mover-trad',hasChanged(['trad']));activate('mover-capital',hasChanged(['k','pool','sarmra','mode']));

    $('delta-target').textContent=`max Δ ${fmtPct(maxDelta)}`;

    $('delta-rank').textContent=moved?`${moved} of ${model.rows.length} rank positions changed`:'No rank positions changed';

    $('mode-note').innerHTML=state.mode==='ALLOC'

      ?`<strong>ALLOC mode is zero-sum</strong>The net-asset pool is fixed at ${fmtNum(state.pool)}. If one block’s required capital rises, it receives a larger share of the pool; the other blocks’ allocated NA falls.`

      :`<strong>TARGET mode expands or contracts capital</strong>Each block receives RC × k. A rise in risk or the target ratio increases total allocated capital; it is not a fixed-pool redistribution.`;

  }

  function diagnostics(base){

    const scenarios=[

      {name:'Management / control capital',counter:'Set Q to 0',change:x=>{x.sarmra=80;}},

      {name:'Risk correlations',counter:'Set all correlations to 0',change:x=>{x.im=0;x.ic=0;x.mc=0;}},

      {name:'Tax gross-up',counter:'Use after-tax ROE directly',change:x=>{x.tax=0;}},

    ];

    const baseRanks=rankMap(base);

    $('diagnostics').innerHTML=scenarios.map(s=>{

      const variant=clone(state);s.change(variant);const tested=calculate(variant);

      const largest=Math.max(...base.rows.map((r,i)=>Math.abs(r.target-tested.rows[i].target)));

      const rankChanges=base.rows.filter(r=>baseRanks[r.name]!==rankMap(tested)[r.name]).length;

      const simple=largest<.0005;

      return `<tr><td><strong>${s.name}</strong></td><td>${s.counter}</td><td>${fmtPct(largest)}</td><td>${rankChanges} of ${base.rows.length}</td><td><span class="tag ${simple?'simplify':'keep'}">${simple?'Potentially simplify':'Keep in model'}</span></td></tr>`;

    }).join('');

  }

  function render(){

    const model=calculate();

    $('q-output').textContent=fmtNum(model.q,3);

    $('pretax-output').textContent=fmtPct(model.preTax);

    [['matrix-im',state.im],['matrix-im-2',state.im],['matrix-ic',state.ic],['matrix-ic-2',state.ic],['matrix-mc',state.mc],['matrix-mc-2',state.mc]].forEach(([id,v])=>$(id).textContent=fmtNum(v,2));

    renderRows(model);

    const allocationLabel=state.mode==='ALLOC'?'Allocated NA equals actual pool':'Allocated NA equals target capital';

    $('reconciliation').innerHTML=`<dt>Total required capital (ΣRC)</dt><dd>${fmtNum(model.totalRC)}</dd><dt>Target capital total (k × ΣRC)</dt><dd>${fmtNum(model.targetTotal)}</dd><dt>Allocated NA (ΣNA<sub>i</sub>)</dt><dd>${fmtNum(model.totalNA)}</dd><dt>Actual net-asset pool</dt><dd>${fmtNum(state.pool)}</dd><dt>Free surplus (pool − target capital)</dt><dd class="${model.freeSurplus<0?'negative':''}">${fmtNum(model.freeSurplus)}</dd><dt>Mode check</dt><dd>${allocationLabel}</dd>`;

    const max=Math.max(.0001,...model.rows.map(r=>r.cost));

    $('bar-chart').innerHTML=model.rows.map(r=>`<div class="bar-group"><div class="bar" style="height:${Math.max(4,r.cost/max*126)}px"><span>${fmtPct(r.cost)}</span></div><div>${escapeHtml(r.name)}</div></div>`).join('');

    diagnostics(model);

    renderMovementMap(model);

    try{localStorage.setItem('cross-allocation-model',JSON.stringify(state));}catch(e){}

  }

  function bind(){

    document.querySelectorAll('#sarmra,#k,#roe,#tax,#trad,#pool,#mode,#corr-im,#corr-ic,#corr-mc').forEach(el=>el.addEventListener('input',()=>{updateStateFromInputs();render();}));

    $('blocks').addEventListener('change',event=>{const t=event.target;if(!t.matches('input[data-row]'))return;const row=state.blocks[+t.dataset.row];row[t.dataset.key]=t.dataset.key==='name'?t.value:num(t.value);render();});

    $('blocks').addEventListener('click',event=>{const b=event.target.closest('[data-delete]');if(!b)return;if(state.blocks.length>1){state.blocks.splice(+b.dataset.delete,1);render();}});

    $('add-row').addEventListener('click',()=>{const n=state.blocks.length+1;state.blocks.push({name:`Block ${n}`,l:100,c:.03,tvog:0,expense:0,strain:0,underwriting:0,alm:0,ins:1,mkt:1,credit:1,other:0});render();});

    $('reset').addEventListener('click',()=>{state=clone(defaults);baseline=clone(defaults);assignInputs();render();});

    $('set-baseline').addEventListener('click',()=>{baseline=clone(state);render();});

    $('download').addEventListener('click',()=>download(calculate()));

  }
